import networkx as nx

def nu2(x):

    while x % 2 == 0:

def build_tree(max_nodes=300, max_depth=8):

    G.add_node(1, depth=0)
    frontier = [1]
    visited = {1}
    for depth in range(max_depth):

                G.add_edge(n, m)
                nxt.append(n)
        frontier = nxt
        if not frontier:
            break
    return G
